# 이 파일은 .github/workflows/update-data.yml 의 수정본입니다.
# GitHub App 에 workflows 권한이 없어 이 작업 환경에서는 .github/ 아래 파일을 커밋할 수 없습니다.
# workflows 권한이 있는 계정/앱으로 아래 내용을 .github/workflows/update-data.yml 에 덮어써 주세요.
# (기존 파일은 들여쓰기가 깨져 매 실행이 실패하고, 매일 06:10 KST 자동 수집이 돌지 않습니다.)

name: update-data

on:
  schedule:
    - cron: "10 21 * * *"   # 06:10 KST
  workflow_dispatch:

permissions:
  contents: write

jobs:
  collect:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"
      - name: 수집 (홈페이지 크롤링)
        env:
          DATA_GO_KR_KEY: ${{ secrets.DATA_GO_KR_KEY }}
        run: |
          python3 scripts/fetch_contracts.py || true
          python3 scripts/detect_anomalies.py
      - name: 커밋
        run: |
          git config user.name "cartelwatch-bot"
          git config user.email "actions@users.noreply.github.com"
          git add data
          if git diff --quiet --cached; then
            echo "변경 없음"
          else
            git commit -m "data: daily update $(date -u +%Y-%m-%d)"
            git push
          fi
